def parse_time_to_minutes(time_text):
    
    first_split = time_text.split("-")
    if len(first_split) != 2:
        return None
    second_split = first_split[0].split(":")
    if len(second_split) != 2:
        return None
    third_split = first_split[1].split(":")
    if len(third_split) != 2:
        return None
    for thing in second_split:
        if not thing.isdigit():
            return None
    for thing2 in third_split:
        if not thing2.isdigit():
            return None
    if second_split[0] > third_split[0]:
        return None
    #seconds_from_second = (second_split[0] * 60) + second_split[1]
    #seconds_from_third = (third_split[0] * 60) + third_split[1]
    #print(second_split, third_split)
    hours1 = "".join(second_split)
    hours2 = "".join(third_split)
    hours1_1 = hours1[:2]
    if hours1_1 > "24":
        return None
    hours1_2 = hours1[-2:]
    total_hours1 = int(hours1_1) * 60 + int(hours1_2)
    hours2_1 = hours2[:2]
    if hours2_1 > "24":
        return None
    hours2_2 = hours2[-2:]
    total_hours2 = int(hours2_1) * 60 + int(hours2_2)
    if total_hours1 == total_hours2:
        return None
    total_minutes = total_hours2 - total_hours1
    #total_hours = int(hours2) - int(hours1)
    #if len(str(total_hours)) == 2:
    #    hour_str = "00" + str(total_hours)
    #elif len(str(total_hours)) == 3:
    #    hour_str = "0" + str(total_hours)
    #else:
    #hour_str = str(total_hours)
    #hour = hour_str[0]
    #minutes = hour_str[-2:]

    #total_minutes = int(hour) * 60 + int(minutes)

    return total_minutes

    
def summarize_schedule(entries):
    
    summary = []
    busiest = ""
    something = {}

    for entry in entries:
        parts = entry.split(",")
        if len(parts) != 3:
            continue
        day = parts[0]
        time = parts[1]
        task = parts[2]

        if day == "" or task == "":
            continue
        parsed_time = parse_time_to_minutes(time)
        if parsed_time == None:
            continue
            
        if day not in something:
            something[day] = {}
        if task not in something[day]:
            something[day][task] = 0
        something[day][task] += parsed_time

    #total_time = 0
    total_per_task = {}
    for whatever in something:
        total_time = 0
        for bleh in something[whatever]:
            total_time += something[whatever][bleh]
            if bleh not in total_per_task:
                total_per_task[bleh] = 0
            total_per_task[bleh] += something[whatever][bleh]
        summary.append(f"{whatever}: {total_time}")

    highest_so_far = 0
    for theloveofgod in total_per_task:
        if total_per_task[theloveofgod] > highest_so_far:
            highest_so_far = total_per_task[theloveofgod]
            #if theloveofgod > busiest:
            #    busiest = theloveofgod
            busiest = theloveofgod
        elif total_per_task[theloveofgod] == highest_so_far:
            if theloveofgod < busiest:
                busiest = theloveofgod
        #print(parsed_time)
    #print(something)
    #print(total_time)
    #print(summary)
    print(total_per_task)

    sorted_summary = list(sorted(summary))

    return sorted_summary, busiest
